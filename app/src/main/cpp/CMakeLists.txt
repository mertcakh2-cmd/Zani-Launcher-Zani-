cmake_minimum_required(VERSION 3.22.1)
project(zani_native LANGUAGES C CXX)

add_library(zani_native SHARED native_bridge.cpp native_util.c)
target_compile_features(zani_native PRIVATE cxx_std_17)
target_compile_options(zani_native PRIVATE -Wall -Wextra)

# Rust remains optional for a plain Android Studio/host CMake build. Run
# scripts/build-rust-android.sh to produce the per-ABI static libraries.
if(ANDROID)
    if(ANDROID_ABI STREQUAL "arm64-v8a")
        set(ZANI_RUST_TARGET "aarch64-linux-android")
    elseif(ANDROID_ABI STREQUAL "armeabi-v7a")
        set(ZANI_RUST_TARGET "armv7-linux-androideabi")
    elseif(ANDROID_ABI STREQUAL "x86_64")
        set(ZANI_RUST_TARGET "x86_64-linux-android")
    endif()
    set(ZANI_RUST_ARCHIVE "${CMAKE_CURRENT_LIST_DIR}/../../../../native/rust/target/${ZANI_RUST_TARGET}/release/${CMAKE_STATIC_LIBRARY_PREFIX}zani_core${CMAKE_STATIC_LIBRARY_SUFFIX}")
    if(EXISTS "${ZANI_RUST_ARCHIVE}")
        target_link_libraries(zani_native PRIVATE "${ZANI_RUST_ARCHIVE}" log android dl m)
        message(STATUS "Zani Rust core enabled: ${ZANI_RUST_TARGET}")
    else()
        message(STATUS "Zani Rust archive not found; using the C validation fallback")
    endif()
endif()
